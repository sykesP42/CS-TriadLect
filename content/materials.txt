# ============================================================================
#  材质表 —— 数媒组工作室
# ----------------------------------------------------------------------------
#  改这里，存盘，回到游戏里按 R，立刻生效：不用重启，不用重新编译。
#
#  这个文件是「覆盖式」的：写了哪一项就改哪一项，没写的保持代码里的默认值。
#  所以想试什么，找到那一行直接改数字就行，不用全看懂。
#
#  参数含义：
#    albedo     基础色（线性空间，不是屏幕上的颜色）。三个数 = 红 绿 蓝。
#               0 全黑、1 全亮。想要蓝一点：把第三个加大、第一个减小。
#    roughness  粗糙度。0 = 镜子（清晰的高光和倒影），1 = 完全漫反射（哑光）。
#    metallic   金属度。1 = 金属（反射环境色，albedo 更像"金属本色"），
#               0 = 塑料 / 木头 / 陶土 / 石头这类非金属。
#    emissive   自发光。物体自己发出的光，给 0 就是不发光。
#    uvscale    贴图平铺次数，只对带贴图的材质有用（比如地板的方格砖）。
#
#  上手就试（改完存盘 → 切回游戏按 R）：
#    · plastic 的 albedo 改成 0.15 0.35 0.85   → 红塑料球变蓝
#    · chrome  的 roughness 改成 0.45          → 镜面球变成磨砂金属
#    · clay    的 roughness 改成 0.12          → 陶土球变成上了釉的陶瓷
#    · floor   的 uvscale 改成 2 2             → 地砖变成 2×2 米的大砖
#    · lamp    的 emissive 改成 12 11 9        → 吊灯变亮，整个房间跟着亮
#  改坏了也不怕：文件里任何一处语法错误都会在控制台报出「哪一行、错在哪」，
#  而且那一次改动会被整个忽略 —— 画面保持上一版的样子，不会黑屏。
# ============================================================================

# ---- 房间本身 --------------------------------------------------------------
material floor {                    # 地板（自带方格砖贴图，albedo 是给贴图乘的颜色）
    albedo    1.00 1.00 1.00
    roughness 0.30
    uvscale   6 6
}

material wall {                     # 四面墙
    albedo    0.30 0.32 0.36
    roughness 0.88
}

material ceiling {                  # 天花板
    albedo    0.16 0.17 0.20
    roughness 0.95
}

# ---- 家具 ------------------------------------------------------------------
material wood {                     # 桌子 / 书柜 / 木箱
    albedo    0.52 0.33 0.16
    roughness 0.55
}

material stone {                    # 终端外壳 / 键盘 / 展台
    albedo    0.52 0.53 0.55
    roughness 0.62
}

material board {                    # 墙上的展板
    albedo    0.68 0.66 0.60
    roughness 0.72
}

# ---- 展台上的三个球（第 2 关「材质」的主角）--------------------------------
material chrome {                   # 镜面球：金属 + 极低粗糙度
    albedo    0.95 0.93 0.90
    roughness 0.06
    metallic  1.0
}

material plastic {                  # 塑料球：红色，非金属，有点亮
    albedo    0.82 0.13 0.11
    roughness 0.26
}

material clay {                     # 陶土球：粗糙的非金属，完全没有高光
    albedo    0.74 0.53 0.32
    roughness 0.92
}

# ---- 发光的两个东西 --------------------------------------------------------
# 它们不只是"看着亮"：吊灯的 emissive 和场景里的主光是同一个位置，
# 屏幕的青色光会把人从黑暗里引到终端面前。
material lamp {                     # 吊灯的灯罩
    albedo    0.00 0.00 0.00
    roughness 0.50
    emissive  4.20 3.80 3.00
}

material screen {                   # 终端屏幕
    albedo    0.00 0.00 0.00
    roughness 0.40
    emissive  0.35 1.35 1.05
}
