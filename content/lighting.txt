# ============================================================================
#  光照 —— 数媒组工作室
# ----------------------------------------------------------------------------
#  改这里，存盘，回到游戏里按 R，立刻生效。
#
#  三行环境色：
#    ambient  全场底光。它不来自任何一盏灯，是"整个世界最暗的地方有多亮"。
#             调到 0 就是全黑 —— 第 1 关「第一束光」就是从 0 开始的。
#    sky      从上往下的环境光颜色（天花板反弹下来的光）。
#    ground   从下往上的环境光颜色（地板反弹上去的光）。
#             这两个是廉价地替代一张真正的 HDR 环境贴图：
#             金属球之所以看起来像金属，靠的就是它反射出了天光和地面光。
#
#  灯块（intensity、color、radius）：
#    color     光的颜色。三个数 = 红 绿 蓝，可以超过 1（那会更亮、更偏色）。
#    intensity 强度。光的亮度按距离平方衰减，所以调大一点很管用。
#    radius    灯的物理尺寸（米）。0.05 是一颗灯泡（高光是个小点），
#              1.0 是一整块发光面板（高光是柔和的一大片）。
#
#  上手就试（改完存盘 → 切回游戏按 R）：
#    · 把 ambient 改成 0 0 0 再按 R   → 房间立刻变黑
#    · 把 light 0 的 intensity 改成 0 → 关掉主光，只剩那盏冷色补光
#    · 把 light 1 的 color 改成 1 0.4 0.1 → 补光变成暖橙色
# ============================================================================

# ---- 环境 ----------------------------------------------------------------
ambient 0.42 0.44 0.50
sky     0.46 0.56 0.78
ground  0.24 0.20 0.17

# ---- 灯 ------------------------------------------------------------------
light 0 {                       # 主光：暖色吊灯，在右后上方（= 吊灯本体的位置）
    color     1.00 0.93 0.82
    intensity 48
    radius    0.9
}

light 1 {                       # 补光：冷色，从左边打过来，把暗部从纯黑里拉回来
    color     0.45 0.62 1.00
    intensity 26
    radius    0.45
}
